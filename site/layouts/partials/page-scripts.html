{{- /* page-scripts.html -- page-scoped JS entries and component CSS. Some pages (customer portal, admin dashboards, communication preferences) load their own component scripts and stylesheets in addition to the main bundle. Those were referenced with hardcoded `/js/...` and `/css/...` paths inside markdown content, which no build step ever produced -- so each of those pages loaded nothing. Pages now declare what they need in front matter and this partial bundles and fingerprints it through the same Hugo Pipes pipeline as everything else. Front matter: scripts: # paths relative to assets/js/ - customer-portal.js styles: # paths relative to assets/css/ - components/communication-preferences.css A page can also pull in the matching component stylesheet automatically by naming the script's directory (e.g. components/x.js -> components/x.css), which is why `styles` is usually unnecessary. Both .css and .scss are accepted; .scss is compiled with libsass first. Stylesheets are emitted through partials/stylesheet.html, the same code path the global bundle uses. NOTE: a *section* stylesheet belongs in the section layout's `headStyles` block, not here. This is for the one-off additions a single page makes. A stylesheet in front matter is silently absent if a future page forgets it. */ -}} {{- $page := . -}} {{- /* --- component stylesheets ---------------------------------------- */ -}} {{- $styles := slice -}} {{- with $page.Params.styles -}} {{- range . -}}{{- $styles = $styles | append . -}}{{- end -}} {{- end -}} {{- /* Derive styles from script names when a matching component CSS exists. */ -}} {{- with $page.Params.scripts -}} {{- range . -}} {{- $dir := path.Dir . -}} {{- $stem := path.BaseName . -}} {{- if ne $dir "." -}} {{- $candidate := printf "css/%s/%s.css" $dir $stem -}} {{- if fileExists (printf "assets/%s" $candidate) -}} {{- if not (in $styles (strings.TrimPrefix "css/" $candidate)) -}} {{- $styles = $styles | append (strings.TrimPrefix "css/" $candidate) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- range $styles -}} {{- partial "stylesheet.html" . -}} {{- end -}} {{- /* --- page-scoped scripts ------------------------------------------ */ -}} {{- with $page.Params.scripts -}} {{- range . -}} {{- partial "entry.html" (dict "entry" . "page" $page) -}} {{- end -}} {{- end -}}
